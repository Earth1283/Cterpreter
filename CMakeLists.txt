cmake_minimum_required(VERSION 3.20)
project(Cterpreter C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(CT_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(CT_FUZZER "Build the fuzz target with libFuzzer instead of the generated corpus" OFF)
option(CT_REPRODUCIBLE "Strip build paths from the output so that builds are reproducible" ON)
option(CT_REPRODUCIBLE_TEST "Verify reproducibility by building the core library twice" OFF)

add_library(cterpreter_core src/lexer.c src/parser.c src/interpreter.c src/memory.c src/builtins.c src/preprocessor.c src/types.c src/optimizer.c)
target_include_directories(cterpreter_core PUBLIC include PRIVATE src)

add_executable(Cterpreter main.c src/terminal.c src/boot.c src/config.c)
target_include_directories(Cterpreter PRIVATE src)
target_link_libraries(Cterpreter PRIVATE cterpreter_core)

if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target cterpreter_core Cterpreter)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(CT_REPRODUCIBLE)
            target_compile_options(${target} PRIVATE
                -ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=. -frandom-seed=${target})
        endif()
        if(CT_SANITIZERS)
            target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PUBLIC -fsanitize=address,undefined)
        endif()
    endforeach()
    target_link_libraries(cterpreter_core PUBLIC m)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND AND UNIX)
        add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py $<TARGET_FILE:Cterpreter>)
    endif()
    add_executable(core_smoke tests/core_smoke.c)
    target_link_libraries(core_smoke PRIVATE cterpreter_core)
    add_test(NAME core_smoke COMMAND core_smoke)
    add_executable(lexer_parser_tests tests/lexer_parser.c)
    target_include_directories(lexer_parser_tests PRIVATE src)
    target_link_libraries(lexer_parser_tests PRIVATE cterpreter_core)
    add_test(NAME lexer_parser COMMAND lexer_parser_tests)
    add_executable(arithmetic_tests tests/arithmetic.c)
    target_link_libraries(arithmetic_tests PRIVATE cterpreter_core)
    add_test(NAME arithmetic COMMAND arithmetic_tests)
    add_executable(varargs_tests tests/varargs.c)
    target_link_libraries(varargs_tests PRIVATE cterpreter_core)
    add_test(NAME varargs COMMAND varargs_tests)
    add_executable(fuzz_tests tests/fuzz.c)
    target_link_libraries(fuzz_tests PRIVATE cterpreter_core)
    if(CT_FUZZER)
        target_compile_definitions(fuzz_tests PRIVATE CT_LIBFUZZER)
        target_compile_options(fuzz_tests PRIVATE -fsanitize=fuzzer,address)
        target_link_options(fuzz_tests PRIVATE -fsanitize=fuzzer,address)
    else()
        add_test(NAME fuzz COMMAND fuzz_tests 2000 1)
    endif()
    if(CT_REPRODUCIBLE AND CT_REPRODUCIBLE_TEST)
        add_test(NAME reproducible_build
                 COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/tests/reproducible.sh
                         ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_GENERATOR})
    endif()
    add_test(NAME reference_programs
             COMMAND ${CMAKE_COMMAND} -E env CTERPRETER=$<TARGET_FILE:Cterpreter>
                     ${CMAKE_CURRENT_SOURCE_DIR}/tests/reference.sh ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME hello_program COMMAND Cterpreter ${CMAKE_CURRENT_SOURCE_DIR}/examples/basics/hello.c Ada)
    set_tests_properties(hello_program PROPERTIES PASS_REGULAR_EXPRESSION "Hello, Ada!")
    add_test(NAME memory_program COMMAND Cterpreter ${CMAKE_CURRENT_SOURCE_DIR}/examples/memory/squares.c)
    set_tests_properties(memory_program PROPERTIES PASS_REGULAR_EXPRESSION "0, 1, 4, 9, 16")
endif()
