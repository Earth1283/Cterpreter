cmake_minimum_required(VERSION 3.20)
project(Cterpreter C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(CT_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(cterpreter_core src/lexer.c src/parser.c src/interpreter.c src/memory.c src/builtins.c src/preprocessor.c)
target_include_directories(cterpreter_core PUBLIC include PRIVATE src)

add_executable(Cterpreter main.c src/terminal.c)
target_include_directories(Cterpreter PRIVATE src)
target_link_libraries(Cterpreter PRIVATE cterpreter_core)

if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target cterpreter_core Cterpreter)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(CT_SANITIZERS)
            target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PUBLIC -fsanitize=address,undefined)
        endif()
    endforeach()
    target_link_libraries(cterpreter_core PUBLIC m)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(core_smoke tests/core_smoke.c)
    target_link_libraries(core_smoke PRIVATE cterpreter_core)
    add_test(NAME core_smoke COMMAND core_smoke)
    add_test(NAME hello_program COMMAND Cterpreter ${CMAKE_CURRENT_SOURCE_DIR}/examples/basics/hello.c Ada)
    set_tests_properties(hello_program PROPERTIES PASS_REGULAR_EXPRESSION "Hello, Ada!")
    add_test(NAME memory_program COMMAND Cterpreter ${CMAKE_CURRENT_SOURCE_DIR}/examples/memory/squares.c)
    set_tests_properties(memory_program PROPERTIES PASS_REGULAR_EXPRESSION "0, 1, 4, 9, 16")
endif()
